# 查找

## 三分

三分查找是一种在 **单峰函数** （先增后减或先减后增）上寻找极值点的算法。它的时间复杂度是 `O(log n)`，适用于浮点数或整数范围上的搜索。

二分查找需要函数为线性的，三分可以解决一些非线性的问题，如函数模型为二次函数类型的

三分法的精髓在于，**每次通过比较两个值的大小，缩小三分之一的区间。直到最后区间的范围小于我们设置的阈值为止**

```c++
#include <iostream>
#include <cmath>

// 设置精度
const double EPS = 1e-8;

// 示例目标函数（你可以替换为你自己的函数）
double f(double x) {
    return -x * x + 4 * x; // 这是一个开口向下的抛物线，最大值在 x=2 处
}

// 三分查找函数（找最大值）
double ternary_search(double left, double right) {
    while (right - left > EPS) {
        double mid1 = left + (right - left) / 3;
        double mid2 = right - (right - left) / 3;

        double f1 = f(mid1);
        double f2 = f(mid2);

        if (f1 < f2) {
            left = mid1;
        } else {
            right = mid2;
        }
    }
    return (left + right) / 2; // 返回近似最优解
}

int main() {
    double left = 0.0;
    double right = 5.0;

    double max_x = ternary_search(left, right);
    std::cout << "Maximum occurs at x = " << max_x << std::endl;
    std::cout << "Maximum value is f(x) = " << f(max_x) << std::endl;

    return 0;
}
```

```c++
}
```

```c++
int ne[N], ans[N], cnt;
void get_next(char s[]) {
	int i = 1, idx = 0, len = strlen(s);
	ne[0] = 0;
	while (i < len) {
		while (s[i] != s[idx] && idx) {

void kmp(char s[], char a[]) {
	int i = 0, j = 0, lens = strlen(s), lena = strlen(a);
	while (i < lens) {
		if (s[i] == a[j]) {
			i++; j++;
		}
		else {
			if (j > 0) {
				j = ne[j - 1];
			}
			else {
				i++;
			}
		}
		if (j == lena) {
			ans[cnt++] = i - lena + 1;
			j = ne[j - 1];
		}
	}
}
```

## 字符串哈希

核心思想：将字符串看成P进制数，P的经验值是131或13331，取这两个值的冲突概率低
小技巧：取模的数用2^64，这样直接用unsigned long long存储，溢出的结果就是取模的结果

```C++
#include <iostream>
#include <string>
#include <vector>

using namespace std;

const int P = 131;

// 初始化哈希表的前缀和数组h和幂次数组p
void init_hasher(const string &s, vector<unsigned long long> &h, vector<unsigned long long> &p) {
    int n = s.size();
    h.resize(n + 1);
    p.resize(n + 1);
    p[0] = 1;
    for (int i = 1; i <= n; ++i) {
        h[i] = h[i - 1] * P + s[i - 1];  // 注意字符从s[0]开始取
        p[i] = p[i - 1] * P;
    }
}

// 获取子串s[l..r]的哈希值（1-based索引）
unsigned long long get_hash(const vector<unsigned long long>& h, const vector<unsigned long long>& p, int l, int r) {
    return h[r] - h[l - 1] * p[r - l + 1];
}

// 比较两个子串是否相同
bool compare_substrings(const vector<unsigned long long>& h, const vector<unsigned long long>& p, 
                        int l1, int r1, int l2, int r2) {
    return get_hash(h, p, l1, r1) == get_hash(h, p, l2, r2);
}

int main() {
    int n, m;
    cin >> n >> m;
    string s;
    cin >> s;
    
    // 初始化哈希数组
    vector<unsigned long long> h, p;
    init_hasher(s, h, p);
    
    // 处理查询
    while (m--) {
        int l1, r1, l2, r2;
        cin >> l1 >> r1 >> l2 >> r2;
        if (compare_substrings(h, p, l1, r1, l2, r2)) {
            cout << "Yes" << endl;
        } else {
            cout << "No" << endl;
        }
    }
    
    return 0;
}
```

# 最短路

## dijkstra

```c++
#include <iostream>
#include <vector>
#include <queue>
#include <climits>
using namespace std;

const int INF = INT_MAX; // 表示无穷大

int main() {
    int n, m;
    cout << "输入节点数和边数: ";
    cin >> n >> m;

    // 使用vector构建邻接表，adj[u]存储所有从u出发的边 (v, w)
    vector<vector<pair<int, int>>> adj(n);

    cout << "输入" << m << "条边（格式：起点 终点 权重）:" << endl;
    for (int i = 0; i < m; ++i) {
        int u, v, w;
        cin >> u >> v >> w;
        adj[u].emplace_back(v, w);  // 有向图
        // 如果是无向图，需添加反向边：adj[v].emplace_back(u, w);
    }

    int start;
    cin >> start;

    vector<int> dist(n, INF);  // 存储从起点到各节点的最短距离
    dist[start] = 0;

    // 优先队列：存储(距离, 节点)，按距离从小到大排序
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;
    pq.push({0, start});

    while (!pq.empty()) {
        int u = pq.top().second;
        int current_dist = pq.top().first;
        pq.pop();

        // 如果当前距离大于记录的距离，说明已找到更优解，跳过
        if (current_dist > dist[u]) continue;

        // 遍历所有邻接节点
        for (auto &edge : adj[u]) {
            int v = edge.first;
            int w = edge.second;

            // 松弛操作
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                pq.push({dist[v], v});
            }
        }
    }

    // 输出结果
    cout << "\n从节点" << start << "出发的最短路径：" << endl;
    for (int i = 0; i < n; ++i) {
        if (dist[i] == INF)
            cout << "节点 " << i << " 不可达" << endl;
        else
            cout << "到节点 " << i << " 的最短距离: " << dist[i] << endl;
    }

    return 0;
}
```

## spfa 算法

### 找最短路

O(m) O(n*m)

```C++
int n;      // 总点数
int h[N], w[N], e[N], ne[N], idx;       // 邻接表存储所有边
int dist[N];        // 存储每个点到1号点的最短距离
bool st[N];     // 存储每个点是否在队列中

// 求1号点到n号点的最短路距离，如果从1号点无法走到n号点则返回-1
int spfa()
{
    memset(dist, 0x3f, sizeof dist);
    dist[1] = 0;
    
    queue<int> q;
    q.push(1);
    st[1] = true;
    
    while (q.size())
    {
        auto t = q.front();
        q.pop();
    
        st[t] = false;
    
        for (int i = h[t]; i != -1; i = ne[i])
        {
            int j = e[i];
            if (dist[j] > dist[t] + w[i])
            {
                dist[j] = dist[t] + w[i];
                if (!st[j])     // 如果队列中已存在j，则不需要将j重复插入
                {
                    q.push(j);
                    st[j] = true;
                }
            }
        }
    }
    
    if (dist[n] == 0x3f3f3f3f) return -1;
    return dist[n];

}
```

### 判负环

```c++
int n;      // 总点数
int h[N], w[N], e[N], ne[N], idx;       // 邻接表存储所有边
int dist[N], cnt[N];        // dist[x]存储1号点到x的最短距离，cnt[x]存储1到x的最短路中经过的点数
bool st[N];     // 存储每个点是否在队列中

// 如果存在负环，则返回true，否则返回false。
bool spfa()
{
    // 不需要初始化dist数组
    // 原理：如果某条最短路径上有n个点（除了自己），那么加上自己之后一共有n+1个点，由抽屉原理一定有两个点相同，所以存在环。

    queue<int> q;
    for (int i = 1; i <= n; i ++ )
    {
        q.push(i);
        st[i] = true;
    }

    while (q.size())
    {
        auto t = q.front();
        q.pop();

        st[t] = false;

        for (int i = h[t]; i != -1; i = ne[i])
        {
            int j = e[i];
            if (dist[j] > dist[t] + w[i])
            {
                dist[j] = dist[t] + w[i];
                cnt[j] = cnt[t] + 1;
                if (cnt[j] >= n) return true;       // 如果从1号点到x的最短路中包含至少n个点（不包括自己），则说明存在环
                if (!st[j])
                {
                    q.push(j);
                    st[j] = true;
                }
            }
        }
```

```c++
#include <iostream>
#include <vector>
#include <queue>
#include <climits>
using namespace std;

const int INF = INT_MAX; // 表示无穷大

int primMST(vector<vector<pair<int, int>>>& graph, int n) {
    vector<int> key(n, INF);    // 存储节点到MST的最小边权
    vector<bool> inMST(n, false); // 标记节点是否已加入MST
    priority_queue<pair<int, int>, vector<pair<int, int>>, greater<pair<int, int>>> pq;

    int start = 0; // 从节点0开始构建MST（可任意选择）
    key[start] = 0;
    pq.push({0, start});

    int totalWeight = 0; // MST的总权重

    while (!pq.empty()) {
        int u = pq.top().second;
        int weight = pq.top().first;
        pq.pop();
        // 如果节点已加入MST，跳过
        if (inMST[u]) continue;

        inMST[u] = true;
        totalWeight += weight;

        // 遍历所有邻接节点
        for (auto &neighbor : graph[u]) {
            int v = neighbor.first;
            int w = neighbor.second;

            // 如果v未加入MST，且发现更小的边权
            if (!inMST[v] && w < key[v]) {
                key[v] = w;
                pq.push({key[v], v});
            }
        }
    }

    // 检查图是否连通
    for (bool visited : inMST) {
        if (!visited) {
            cout << "图不连通，无法生成MST！" << endl;
            return -1;
        }
    }

    return totalWeight;
}

int main() {
    int n, m;
    cout << "输入节点数和边数: ";
    cin >> n >> m;

    // 邻接表：graph[u]存储{u的邻居v, 边权w}
    vector<vector<pair<int, int>>> graph(n);

    cout << "输入" << m << "条边（格式：起点 终点 权重）:" << endl;
    for (int i = 0; i < m; ++i) {
        int u, v, w;
        cin >> u >> v >> w;
        graph[u].emplace_back(v, w);
        graph[v].emplace_back(u, w); // 无向图需添加双向边
    }

    int result = primMST(graph, n);
    if (result != -1) {
        cout << "最小生成树的总权重为: " << result << endl;
    }

    return 0;
}
```

### 朴素prime算法

```c++
#define _CRT_SECURE_NO_WARNINGS
#include<iostream>
#include<cstring>

using namespace std;

const int N = 510;
int g[N][N], dist[N];
bool st[N];
int n, m;

int Prim()
{
	int res = 0;
	memset(dist, 0x3f, sizeof dist); // 将所有节点的距离初始化为一个很大的值（无穷大）。
	dist[1] = 0; // 从节点1开始算法。

	for (int i = 0; i < n; ++i) // 遍历所有节点。
	{
		int t = -1;
		// 找到还未加入集合的节点中距离最小的节点。
		for (int j = 1; j <= n; ++j)
		{
			if (!st[j] && (t == -1 || dist[t] > dist[j]))
				t = j;
		}
		st[t] = true; // 标记选中的节点为已访问。

		// 如果最小距离仍保持初始值，说明有些节点是不可达的，图是非连通的。
		if (dist[t] == 0x3f3f3f3f) return 0x3f3f3f3f;
		
		// 累加记得提前，防止负权自环。
		res += dist[t]; // 将当前节点的距离累加到结果中。

		// 更新其他节点到集合的最小距离，如果有更小的距离则更新。
		for (int j = 1; j <= n; ++j)
			dist[j] = min(dist[j], g[t][j]);
	}
	return res; // 返回最小生成树的权值和。
}

int main()
{
	cin.tie(0);
	ios::sync_with_stdio(false);
	memset(g, 0x3f, sizeof g); // 初始化所有边的权值为一个很大的值（无穷大）。
	cin >> n >> m;
	for (int i = 0; i < m; ++i)
	{
		int u, v, w;
		cin >> u >> v >> w;
		g[u][v] = g[v][u] = min(g[u][v], w); // 更新边的权值。
	}
	int t = Prim(); // 执行Prim算法得到最小生成树的权值和。
	if (t == 0x3f3f3f3f) cout << "impossible" << endl; // 如果最小生成树不存在，则输出"impossible"。
	else cout << t << endl; // 输出最小生成树的权值和。
	return 0;
}
```

### 克鲁斯卡尔

```C++
#include <bits/stdc++.h>
using namespace std;
const int N = 1e6;
int fa[N], rk[N];
// 定义边的结构体
struct Edge {
    int u, v, w;
    bool operator < (const Edge& other) const {
        return w < other.w;  // 按权重升序排序
    }
};
int find(int x) {
    return x == fa[x] ? x : fa[x] = find(fa[x]);
}
bool unite(int x, int y) {
    int rx = find(x), ry = find(y);
    if (rx == ry) return false;  // 同集合，无需合并
    // 按秩合并
    if (rk[rx] > rk[ry]) {
        fa[ry] = rx;
    }
    else {
        fa[rx] = ry;
        if (rk[rx] == rk[ry]) {
            rk[ry]++;  // 秩相等时，合并后秩+1
        }
    }
    return true;
};
int main() {
    int n, m;
    cin >> n >> m;  // n个顶点，m条边
    vector<Edge> edges(m);
    for (int i = 0; i < m; ++i) {
        cin >> edges[i].u >> edges[i].v >> edges[i].w;
    }

    // 按权重对边排序
    sort(edges.begin(), edges.end());

    // 并查集初始化
    for (int i = 0; i <= n; ++i) {
        fa[i] = i;  // 初始父节点为自己
    }
    // 按秩合并的Union函数
   
    int res = 0, cnt = 0;  // res存储总权重，cnt记录选中的边数
    for (auto& e : edges) {
        if (unite(e.u, e.v)) {
            res += e.w;
            cnt++;
            if (cnt == n - 1) break;  // 已选满n-1条边，提前终止
        }
    }
    // 输出最小生成树的总权重
    cout << res << endl;
    return 0;
}
```

### 克鲁斯卡尔重构树

在克鲁斯卡尔求最短路的基础上对原来的图进行一定的修改，与克鲁斯卡尔求最短路的区别在于每当找到一条树边时，将其变成一个点，这个点连接这条边连接的两个联通块（的根节点），然后就可以得到一棵树，这棵树的[叶子结点](https://so.csdn.net/so/search?q=叶子结点&spm=1001.2101.3001.7020)，其余节点都是树边，权值就是树边的长度。

克鲁斯卡尔重构树通过将 MST 转化为二叉树结构，将路径极值问题转化为 LCA 查询，显著提升了处理效率。其核心作用在于**高效解决与 MST 相关的路径极值查询、动态连通性及 K 小生成树问题**，是图论算法中重要的优化工具。

### **1. 路径极值查询**

```c++
#include <bits/stdc++.h>
using namespace std;
struct Edge {
    int u, v, w;
    bool operator<(const Edge &other) const {
        return w > other.w; // 改为降序排序，构建最大生成树 [[1]]，找最大值
    }
};
struct Edge {
    int u, v, w;
    bool operator<(const Edge &other) const {
        return w < other.w; // 最小生成树按权值升序排序，找最小值
    }
};

const int MAXN = 2e5 + 5; // 原图节点上限
const int LOG = 20; // 倍增深度
int parent[MAXN * 2]; // 并查集父节点
int depth[MAXN * 2]; // 节点深度
int up[MAXN * 2][LOG]; // 倍增数组
int val[MAXN * 2]; // 节点权值
vector<int> adj[MAXN * 2]; // 重构树的邻接表

// 并查集查找
int find(int x) {
    return parent[x] == x ? x : parent[x] = find(parent[x]);
}

// 预处理倍增数组和深度
void dfs(int u, int p) {
    depth[u] = depth[p] + 1;
    up[u][0] = p;
    for (int i = 1; i < LOG; ++i) {
        up[u][i] = up[up[u][i-1]][i-1];
    }
    for (int v : adj[u]) {
        if (v != p) {
            dfs(v, u);
        }
    }
}

// 构建克鲁斯卡尔重构树
int build_kruskal(vector<Edge> &edges, int n) {
    int total_nodes = n;
    sort(edges.begin(), edges.end()); // 按权值排序
    
    // 初始化并查集和父节点
    for (int i = 1; i <= n; ++i) {
        parent[i] = i;
    }
    
    for (auto &e : edges) {
        int u = find(e.u);
        int v = find(e.v);
        if (u != v) {
            // 创建新节点
            val[++total_nodes] = e.w; // 新节点权值为当前边权
            parent[total_nodes] = total_nodes; // 新节点的父节点是自己
            
            // 合并两个集合
            parent[u] = total_nodes;
            parent[v] = total_nodes;
            
            // 建立树结构
            adj[total_nodes].push_back(u);
            adj[total_nodes].push_back(v);
            adj[u].push_back(total_nodes);
            adj[v].push_back(total_nodes);
        }
    }
    
    // 找到根节点（最大节点编号）
    int root = total_nodes;
    while (parent[root] != root) root = parent[root];
    
    // 预处理倍增数组和深度
    memset(up, 0, sizeof(up));
    dfs(root, 0);
    
    return root;
}

// 查询LCA并获取路径极值
int lca(int u, int v) {
    if (depth[u] < depth[v]) swap(u, v);
    // 提升较深节点到同一深度
    for (int i = LOG - 1; i >= 0; --i) {
        if (depth[u] - (1 << i) >= depth[v]) {
            u = up[u][i];
        }
    }
    if (u == v) return u;
    // 同步提升
    for (int i = LOG - 1; i >= 0; --i) {
        if (up[u][i] != up[v][i]) {
            u = up[u][i];
            v = up[v][i];
        }
    }
    return up[u][0];
}

int main() {
    int n, m, q;
    cin >> n >> m;
    vector<Edge> edges(m);
    for (int i = 0; i < m; ++i) {
        cin >> edges[i].u >> edges[i].v >> edges[i].w;
    }
    
    int root = build_kruskal(edges, n); // 构建重构树
    
    cin >> q;
    while (q--) {
        int u, v;
        cin >> u >> v;
        int ancestor = lca(u, v);
        cout << val[ancestor] << endl; // LCA的权值即为路径极值，最大值
    }
    
    return 0;
}
```

### **2. K 小生成树**

```c++
#include <iostream>
#include <vector>
#include <algorithm>
#include <queue>
using namespace std;

const int MAXN = 2e5 + 5;  // 假设节点数最大为1e5，内部节点数同样
const int LOG = 20;

struct Edge {
    int u, v, w;
    bool selected = false;
    bool operator<(const Edge& other) const {
        return w < other.w;
    }
};

int n, m, k;
vector<Edge> edges;
vector<Edge> mst_edges;
int sum_mst;

// 克鲁斯卡尔重构树相关变量
int dsu_parent[MAXN * 2];  // 并查集父节点
int tree_parent[MAXN * 2]; // 树结构的父节点
int val[MAXN * 2];         // 节点权值
int left_child[MAXN * 2], right_child[MAXN * 2];
int cnt;  // 当前节点数

// LCA预处理
int fa[MAXN * 2][LOG];
int depth[MAXN * 2];

void build_mst() {
    vector<int> dsu(n + 1);
    for (int i = 1; i <= n; ++i) dsu[i] = i;

    sort(edges.begin(), edges.end());
    sum_mst = 0;
    mst_edges.clear();

    auto find = [&](int x) {
        while (dsu[x] != x) x = dsu[x];
        return x;
    };

    for (auto& e : edges) {
        int fu = find(e.u);
        int fv = find(e.v);
        if (fu != fv) {
            sum_mst += e.w;
            e.selected = true;
            mst_edges.push_back(e);
            dsu[fu] = fv;
            if (mst_edges.size() == n - 1) break;
        }
    }
}

void build_kruskal_tree() {
    for (int i = 1; i <= 2 * n; ++i) {
        dsu_parent[i] = i;
        tree_parent[i] = 0;
        left_child[i] = right_child[i] = 0;
        val[i] = 0;
    }
    cnt = n;

    auto find = [&](int x) {
        while (dsu_parent[x] != x) x = dsu_parent[x];
        return x;
    };

    sort(mst_edges.begin(), mst_edges.end());
    for (auto& e : mst_edges) {
        int u = e.u, v = e.v, w = e.w;
        int fu = find(u);
        int fv = find(v);
        if (fu != fv) {
            ++cnt;
            val[cnt] = w;
            left_child[cnt] = fu;
            right_child[cnt] = fv;
            tree_parent[fu] = cnt;
            tree_parent[fv] = cnt;
            dsu_parent[fu] = cnt;
            dsu_parent[fv] = cnt;
            dsu_parent[cnt] = cnt;
        }
    }
}

void preprocess_lca() {
    int root = 0;
    for (int i = 1; i <= cnt; ++i) {
        if (tree_parent[i] == 0) {
            root = i;
            break;
        }
    }

    queue<int> q;
    q.push(root);
    depth[root] = 0;
    fa[root][0] = 0;

    while (!q.empty()) {
        int u = q.front();
        q.pop();
        if (left_child[u]) {
            int v = left_child[u];
            depth[v] = depth[u] + 1;
            fa[v][0] = u;
            q.push(v);
        }
        if (right_child[u]) {
            int v = right_child[u];
            depth[v] = depth[u] + 1;
            fa[v][0] = u;
            q.push(v);
        }
    }

    for (int j = 1; j < LOG; ++j) {
        for (int i = 1; i <= cnt; ++i) {
            fa[i][j] = fa[fa[i][j - 1]][j - 1];
        }
    }
}

int get_max_edge(int u, int v) {
    if (depth[u] < depth[v]) swap(u, v);
    for (int k = LOG - 1; k >= 0; --k) {
        if (depth[u] - (1 << k) >= depth[v]) {
            u = fa[u][k];
        }
    }
    if (u == v) return val[u];

    for (int k = LOG - 1; k >= 0; --k) {
        if (fa[u][k] != fa[v][k]) {
            u = fa[u][k];
            v = fa[v][k];
        }
    }
    return val[fa[u][0]];
}

int main() {
    cin >> n >> m >> k;
    for (int i = 0; i < m; ++i) {
        Edge e;
        cin >> e.u >> e.v >> e.w;
        edges.push_back(e);
    }

    build_mst();
    if (mst_edges.size() != n - 1) {
        cout << -1 << endl;
        return 0;
    }

    build_kruskal_tree();
    preprocess_lca();

    vector<int> candidates;
    for (auto& e : edges) {
        if (!e.selected) {
            int max_w = get_max_edge(e.u, e.v);
            if (e.w < max_w) {
                candidates.push_back(sum_mst - max_w + e.w);
            }
        }
    }

    sort(candidates.begin(), candidates.end());

    if (k == 1) {
        cout << sum_mst << endl;
        return 0;
    }
    k--;

    if (candidates.size() < k) {
        cout << -1 << endl;
    } else {
        cout << candidates[k - 1] << endl;
    }

    return 0;
}
```

## Tarjan

### 1.求强连通分量（环）

```C++
#include <bits/stdc++.h>
using namespace std;

const int N = 1e6;
vector<int> g[N];  // 邻接表
int disc[N], low[N], timer = 1;
bool visited[N];
stack<int> s;
vector<vector<int>> cycles;  // 存储所有环

void tarjan(int u, int parent) {
    disc[u] = low[u] = timer++;
    s.push(u);
    visited[u] = true;

    for (int v : g[u]) {
        if (!disc[v]) {
            tarjan(v, u);
            low[u] = min(low[u], low[v]);
        } else if (visited[v] && v != parent) {
            low[u] = min(low[u], disc[v]);
        }
    }

    if (low[u] == disc[u]) {
        vector<int> cycle;
        while (true) {
            int v = s.top(); s.pop();
            visited[v] = false;
            cycle.push_back(v);
            if (v == u) break;
        }
        if (cycle.size() > 1) {  // 过滤单节点自环
            cycles.push_back(cycle);
        }
    }
}

void solve() {
    int n, m;
    cin >> n >> m;
    for (int i = 0; i < m; i++) {
        int x, y;
        cin >> x >> y;
        g[x].push_back(y);
        g[y].push_back(x);
    }
```

## 网络流问题

### 最大流

常见的网络流问题包括但不限于以下类型问题。

- 最大流问题：对于网络 G=(V,E)，给每条边指定流量，得到合适的流f ，使得f 的流量尽可能大。此时我们称f是G的最大流。

- 最小割问题：对于网络 G=(V,E)，找到合适的s-t 割{s,t}，使得{s,t}的容量尽可能小。此时我们称 {s,t} 是G的最小割,最小割和最大流等价。

- 最小费用最大流问题：在网络 G=(V,E) 上，对每条边给定一个权值 W(u,v) ，称为费用（cost），含义是单位流量通过 (u,v)所花费的代价。对于 G 所有可能的最大流，我们称其中总费用最小的一者为最小费用最大流。

    
  
  DINIC算法，时间复杂度*O*(V^2E),求解最大流，当指定了源点 和汇点时，也可以使用这个算法来求解最小割

  ```c++
      Edge forward = {v, (int)graph[v].size(), cap};
      Edge reverse = {u, (int)graph[u].size(), 0};
      graph[u].push_back(forward);
      graph[v].push_back(reverse);

  bool bfs(int s, int t) {
      queue<int> q;
      level.assign(n, -1);
      q.push(s);
      level[s] = 0;
      while (!q.empty()) {
          int u = q.front(); q.pop();
          for (auto &e : graph[u]) {
              if (level[e.to] == -1 && e.capacity > 0) {
                  level[e.to] = level[u] + 1;
                  q.push(e.to);
              }
          }
      }
      return level[t] != -1;
  }
  
  // DFS 多路增广
  int dfs(int u, int t, int flow) {
      if (u == t) return flow;
      for (int &i = ptr[u]; i < graph[u].size(); ++i) {
          Edge &e = graph[u][i];
          if (level[e.to] == level[u] + 1 && e.capacity > 0) {
              int pushed = dfs(e.to, t, min(flow, e.capacity));
              if (pushed > 0) {
                  e.capacity -= pushed;
                  graph[e.to][e.rev].capacity += pushed;
                  return pushed;
              }
          }
      }
      return 0;
  }
  
  // Dinic 算法主体
  int max_flow(int s, int t) {
      int flow = 0;
      while (bfs(s, t)) {  // 循环分层
          ptr.assign(n, 0);
          while (int pushed = dfs(s, t, INF)) {  // 多路增广
              flow += pushed;
          }
      }
      return flow;
  }
  ```

## 直线

将直线表示为参数形式 *x*(*t*)=*x*0+*b*⋅*t*，*y*(*t*)=*y*0−*a*⋅*t*，其中 (*x*0,*y*0) 是直线上的一点。

#### **情况一：b ≠ 0**

当 *b*=0 时，可以令 *x*0=0，代入方程求 *y*0：

*a*⋅0+*b*⋅*y*0+*c*=0⟹*y*0=−*b**c*

因此，选择点 (*x*0,*y*0)=(0,−*b**c*)。

#### **情况二：b = 0**

当 *b*=0 时，直线方程变为 *a**x*+*c*=0，此时 *a*=0（因为题目保证 *a* 和 *b* 不同时为 0），可以令 *y*0=0，代入方程求 *x*0：

*a*⋅*x*0+*c*=0⟹*x*0=−*a**c*

## BSGS算法

来看这样一个数论问题：给定一个质数 p，以及正整数 a，b，求满足同余方程 ![a^x \equiv b (mod \; p)](https://latex.csdn.net/eq?a%5Ex%20%5Cequiv%20b%20%28mod%20%5C%3B%20p%29) 的最小非负整数 ![x](https://latex.csdn.net/eq?x)，无满足的 ![x](https://latex.csdn.net/eq?x) 则输出 -1。那么，可以使用BSGS算法来解决，要求a与p互质,时间复杂度为O(sqrt(m))

```c++
ll bsgs(ll a, ll b, ll mod){
    map<ll, ll> mp;
    ll cur = 1, t = sqrt(mod) + 1;
    for(int B = 1; B <= t; B++){
        cur = cur * a % mod;
        mp[b * cur % mod] = B;
    }
    ll now = cur;
    for(int A = 1; A <= t; A++){
        if(mp[now]) return (ll)A * t - mp[now];
        now = now * cur % mod;
    }
    return -1;
}
```

## Lucas定理——求组合数

（n-1）&i==i 表示组合数C(i,n-i)  为奇数，反之，为偶数。


原理：若p是质数，则对于任意整数 1 <= m <= n，有：
    C(n, m) = C(n % p, m % p) * C(n / p, m / p) (mod p)

- **C(n,m) 为奇数**：当且仅当在二进制表示中，m的所有为 1 的位在 n*n*的对应位也为 1。换句话说，m*m* 的二进制表示是 n*n* 的二进制表示的一个“子集”，即 m AND n=m*m* AND *n*=*m*。
- **C(n,m) 为偶数**：如果存在至少一个 m*m* 的为 1 的位在 n*n* 的对应位为 0。

```C++
int qmi(int a, int k, int p)  // 快速幂模板
{
    int res = 1 % p;
    while (k)
    {
        if (k & 1) res = (LL)res * a % p;
        a = (LL)a * a % p;
        k >>= 1;
    }
    return res;
}

int C(int a, int b, int p)  // 通过定理求组合数C(a, b)
{
    if (a < b) return 0;

    LL x = 1, y = 1;  // x是分子，y是分母
    for (int i = a, j = 1; j <= b; i --, j ++ )
    {
        x = (LL)x * i % p;
        y = (LL) y * j % p;
    }
    
    return x * (LL)qmi(y, p - 2, p) % p;

}

int lucas(LL a, LL b, int p)
{
    if (a < p && b < p) return C(a, b, p);
    return (LL)C(a % p, b % p, p) * lucas(a / p, b / p, p) % p;
}
```

### 扩展：S(n,m) 的奇偶性 

求 S(n,m) 的奇偶性实质就是求 S(n,m)%2

对于第二类斯特林数，首先有：S[i][j]=S[i−1][j−1]+j∗S[i−1][j]

那么在模 2 的情况下 ，有：

当 j 为偶数：S[i][j] ≡（S[i−1][j−1]%2）
当 j 为奇数：S[i][j] ≡（(S[i−1][j−1]+S[i−1][j])%2）
于是，可以倒过来，即有：

当 j 为偶数时：S[i][j] 会被加到 S[i+1][j+1] 和 S[i+1][j] 
当 j 为奇数时：S[i][j] 会被加到 S[i+1][j+1]
于是，令 a=n−m，b=(m+1)/2，则答案就相当于把 a 个相同的球分成 b 组，每组个数可以为 0 的方案总数，即：C(b−1，a+b−1)

## NIM游戏

给定N堆物品，第i堆物品有Ai个。两名玩家轮流行动，每次可以任选一堆，取走任意多个物品，可把一堆取光，但不能不取。取走最后一件物品者获胜。两人都采取最优策略，问先手是否必胜。

我们把这种游戏称为NIM博弈。把游戏过程中面临的状态称为局面。整局游戏第一个行动的称为先手，第二个行动的称为后手。若在某一局面下无论采取何种行动，都会输掉游戏，则称该局面必败。
所谓采取最优策略是指，若在某一局面下存在某种行动，使得行动后对面面临必败局面，则优先采取该行动。同时，这样的局面被称为必胜。我们讨论的博弈问题一般都只考虑理想情况，即两人均无失误，都采取最优策略行动时游戏的结果。
NIM博弈不存在平局，只有先手必胜和先手必败两种情况。

定理： NIM博弈先手必胜，当且仅当 A1 ^ A2 ^ … ^ An != 0

## 反NIM游戏

NIM 游戏：石子堆数的异或(^)和为不为零，先手必胜；

anti_NIM游戏：

(1) 如果全部为1，若为n为偶数，那么先手必胜
(2) 如果存在ai>1，那么如果异或和不为0，那么先手必胜

## SG函数

### 可以用SG函数解决的问题所需要满足的条件：

（1）游戏人数为两人

（2）两人交替进行某种游戏规定的操作，每次操作，选手都可以在当前合法的操作中选取一种

（3）对于游戏的任意一种可能的局面，合法的操作集合只取决于这个局面的本身，而与操作者无关

就拿巴什博弈举例，每个操作者每次都可以拿走1~m块石头，每次拿走石头的个数取决于当前还剩多少块石头，而与之前的操作无关，这就是一个可以利用SG函数来解决的问题。

### 下面来对P-position（必败点）和N-position（必胜点）来作下解释：

处在必胜点的选手如果不出现决策上的失误，则选手是必胜的，同理，处在必败点的选手是必败的。

依旧拿巴什博弈举例，假如一开始给定m+2块石头，每次可以取1~m块石头，则先手必胜，所以m+2也就是一个必胜点，若一开始给定m+1块石头，则无论先手第一次取几块石头，后手都可以一次性取完剩余的石头，所以m+1是一个必败点。

### 必败点和必胜点满足的性质：

（1）终结点是必败点

（2）可以进行一次操作到达必败点的为必胜点

（3）从必败点只能到达必胜点

### mex函数

表示最小的不属于这个集合的非负整数，比如mex{0,1,3,4}=2,mex{1，2，3}=0;

任何一个可以用SG函数解决的问题都可以抽象成一个有向图游戏，SG函数是用于为这种有向图游戏提供最优策略的。（SG(x)=0代表x为必败态，SG（x）!=0代表x为必胜态）

### 对于给定游戏规则下的有向无环图，

### 定义SG（x）=mex{SG(y) | y是x的后继}

容易知道终止状态的SG值一定是0，对于一个x满足SG(x)=0,则对于x的所有后继（经过一次操作可到达的状态）y一定有SG（y）!=0,类似的，对于一个x满足SG(x)！=0,则对于x的所有后继y均有SG（y）==0。

### 顶点SG值的意义：

当SG（x）=k时，表明对于任何一个0<=i<k,都存在x的一个后继y满足SG(y)=i。

### 下面给出常见博弈论的解题方法：

把原问题按照游戏规则的差别分解成多个独立的子游戏，并计算每个子游戏下的SG值，最后求出所有子游戏的SG值的异或

### 最后给出SG值的计算方法：

（1）若可选步数为1~m的连续整数（巴什博弈），则SG（x）=x%(m+1)

（2）若可选步数为任意步，则SG（x）=x

（3）剩余的情况只能具体问题具体分析了，下面给出求SG的函数模板

打表计算SG值

```C++
#include<cstdio>
#include<cstring>
#include<algorithm>
#include<iostream>
#include<queue>
using namespace std;
const int N=1e3+10;//具体数据 
int f[N],sg[N],vis[N];
//f[]记录可以进行的操作（比如取走多少个石子）
//sg[]记录每种状态的sg值
//vis[x]用于标记出现过的x的后继的sg值
void SG(int n)
{
	memset(sg,0,sizeof sg);
	//0是终止状态（必败态），所以sg[0]一定为0 
	for(int i=1;i<=n;i++)
	{
		memset(vis,0,sizeof vis);
		for(int j=1;f[j]<=i;j++)
			vis[sg[i-f[j]]]=1;
		//i-f[j]是状态i进行一次操作可以到达的状态 
		for(int j=0;j<=n;j++)//求解i的后继的sg值中未出现的最小自然数 
		{
			if(!vis[j])
			{
				sg[i]=j;
				break;
			}
		} 
	}
}
```

## LCA

```c++
//链式前向星
int h[NN], ne[NN], e[NN], idx;
int d[N]/*结点深度*/;
int st[N]/*标记是否成树*/;
int dp[N][22]/* dp[i][j] 第i个结点 向上 走( 2^j )深度 到达的结点 */;
int n, m, s;

void add(int x, int y) {
	e[++idx] = y;
	ne[idx] = h[x];
	h[x] = idx;
	return;
}
void dfs(int k) {
	for (int i = 1; (1 << i/* 2^i次方 */) < d[k]; i++) {
		dp[k][i] = dp[dp[k][i - 1]][i - 1];
	}
	for (int i = h[k]; i; i = ne[i]) {
		int j = e[i];
		if (st[j])continue;
		st[j] = 1;
		d[j] = d[k] + 1;
		dp[j][0] = k;//j的父节点是k
		dfs(j);
	}
	return;
}
int lca(int x, int y) {
	/* 保证x深度大于y */
	if (d[x] < d[y])swap(x, y);
	int k = log2(d[x] - d[y]);
	for (int i = k; i >= 0; i--) {
		if (d[dp[x][i]] >= d[y])x = dp[x][i];
	}
	if (x == y)return y;/* x,y走到同一深度，发现两个结点相同，即y是x祖先*/
	k = log2(d[x]);
	for (int i = k; i >= 0; i--) {
		/* x,y向上走(2^i)深度，若到达同一结点就少走点 */
		if (dp[x][i] == dp[y][i])continue;
		x = dp[x][i]; y = dp[y][i];
	}
	return dp[x][0];
}
int main() {
	scanf("%d%d%d", &n, &m, &s);
	for (int i = 1, x, y; i < n; i++) {
		scanf("%d%d", &x, &y);
		add(x, y); add(y, x);
	}

	st[s] = 1;
	d[s] = 1;/* 与d[0]=0相互区别 */
	dfs(s);

	while (m--) {
		int x, y;
		scanf("%d%d", &x, &y);
		printf("%d\n", lca(x, y));
	}
}
```

```C++
// /* 线段树 区间修改 区间查询（lazy标记区间）洛谷P3372 */
#include<iostream>
using namespace std;
const int N = 2e5 + 10;
#define  int long long
struct node {
	int l, r;
	int sum;
	int lz = 0;
}tr[N * 4];
int a[N], n, m;
void build(int u, int l, int r)
{
	if (l == r)tr[u] = { l,r,a[r],0 };
	else
	{
		tr[u] = { l,r,0,0 };
		int mid = (l + r) >> 1;
		build(u << 1, l, mid);
		build(u << 1 | 1, mid + 1, r);
		tr[u].sum = tr[u << 1].sum + tr[u << 1 | 1].sum;
	}
}
void pushdown(int u)
{
	if (tr[u].lz > 0)
	{
		tr[u << 1].lz += tr[u].lz;
		tr[u << 1 | 1].lz += tr[u].lz; 
		int mid = (tr[u].l + tr[u].r) >> 1;
		tr[u << 1].sum += tr[u].lz * (mid - tr[u].l + 1);
		tr[u << 1 | 1].sum += tr[u].lz * (tr[u].r - mid);
		tr[u].lz = 0;
	}
}
void modify(int u, int l, int r, int v)
{
	if (tr[u].l >= l && tr[u].r <= r)
	{
		tr[u].sum += v * (tr[u].r - tr[u].l + 1);
		tr[u].lz += v;
	}
	else
	{
		pushdown(u);//注意
		int mid = (tr[u].l + tr[u].r) >> 1;
		if (l <= mid)modify(u << 1, l, r, v);
		if (r > mid)modify(u << 1 | 1, l, r, v);
		tr[u].sum = tr[u << 1].sum + tr[u << 1 | 1].sum;
	}
}
int query(int u, int l, int r)
{
	
	if (l <= tr[u].l && tr[u].r <= r)return tr[u].sum;
	pushdown(u);
	int mid = (tr[u].l + tr[u].r) >> 1;
	int sum = 0;
	if (l <= mid)sum += query(u << 1, l, r);
	if (r > mid)sum += query(u << 1 | 1, l, r);
	return sum;
}
```

## 主席树

### 查询区间前K大值

```c++
#include <iostream>
#include <vector>
#include <algorithm>
using namespace std;

typedef long long ll;
const int MAXN = 1e5 + 5;

struct Node {
    int l, r;       // 左右子节点编号
    int cnt;        // 当前值域区间内元素个数
    ll sum;         // 当前值域区间内元素和
} tree[MAXN * 20];

vector<int> roots;  // 各个版本的根节点
int tot;            // 节点总数
vector<int> a;      // 原始数组
vector<int> sorted; // 排序后的数组（用于离散化）

// 离散化：获取x的排名（从1开始）
int get_id(int x) {
    return lower_bound(sorted.begin(), sorted.end(), x) - sorted.begin() + 1;
}

// 构建空树
int build(int l, int r) {
    int p = ++tot;
    tree[p].cnt = tree[p].sum = 0;
    if (l == r) return p;
    
    int mid = (l + r) >> 1;
    tree[p].l = build(l, mid);
    tree[p].r = build(mid + 1, r);
    return p;
}

// 更新：在版本root基础上，在pos位置增加val
int update(int root, int l, int r, int pos, int val) {
    int p = ++tot;
    tree[p] = tree[root];
    tree[p].cnt += 1;
    tree[p].sum += val;
    
    if (l == r) return p;
    
    int mid = (l + r) >> 1;
    if (pos <= mid) 
        tree[p].l = update(tree[root].l, l, mid, pos, val);
    else 
        tree[p].r = update(tree[root].r, mid + 1, r, pos, val);
    return p;
}

// 查询前k大的和
ll query(int u, int v, int l, int r, int k) {
    if (l == r) {
        return 1LL * k * sorted[l-1]; // 注意离散化时是从1开始的
    }
    
    int mid = (l + r) >> 1;
    int right_cnt = tree[tree[v].r].cnt - tree[tree[u].r].cnt;
    ll right_sum = tree[tree[v].r].sum - tree[tree[u].r].sum;
    
    if (right_cnt >= k) {
        // 全部在右子树
        return query(tree[u].r, tree[v].r, mid + 1, r, k);
    } else {
        // 右子树全取，剩下的从左子树取
        return right_sum + query(tree[u].l, tree[v].l, l, mid, k - right_cnt);
    }
}

int main() {
    int n, m;
    cin >> n >> m;
    a.resize(n);
    for (int i = 0; i < n; ++i) {
        cin >> a[i];
    }
    
    // 离散化
    sorted = a;
    sort(sorted.begin(), sorted.end());
    sorted.erase(unique(sorted.begin(), sorted.end()), sorted.end());
    int len = sorted.size();
    
    // 初始化
```
